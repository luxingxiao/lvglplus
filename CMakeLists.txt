cmake_minimum_required(VERSION 3.20)
project(lvglplus VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(LVP_WITH_LVGL "Build real LVGL renderer and runner" ON)
option(LVP_SANITIZE "Enable address/undefined behavior sanitizers" OFF)
set(LVP_LVGL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/lvgl" CACHE PATH "LVGL 9.5.0 checkout")
set(LVP_UI_DIR "" CACHE PATH "Generated UI directory (optional)")
set(LVP_UI_NAME "vehicle" CACHE STRING "Generated schema symbol prefix")
if(LVP_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(lvglplus STATIC sdk/src/model.c sdk/src/vm.c sdk/src/can.c)
add_library(LVGLPlus::core ALIAS lvglplus)
target_include_directories(lvglplus PUBLIC sdk/include)
if(NOT MSVC)
  target_link_libraries(lvglplus PUBLIC m)
  target_compile_options(lvglplus PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(lvglplus_package STATIC sdk/src/package.c)
add_library(LVGLPlus::package ALIAS lvglplus_package)
target_link_libraries(lvglplus_package PUBLIC lvglplus)
if(LVP_WITH_LVGL)
  enable_language(CXX)
  if(NOT TARGET lvgl)
  if(NOT EXISTS "${LVP_LVGL_DIR}/lvgl.h")
    message(FATAL_ERROR "LVGL missing. Run pnpm bootstrap, or set LVP_LVGL_DIR.")
  endif()
  set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/sdk/ports/lv_conf.h" CACHE PATH "" FORCE)
  set(LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
  set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
  set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
  add_subdirectory("${LVP_LVGL_DIR}" "${CMAKE_BINARY_DIR}/lvgl" EXCLUDE_FROM_ALL)
  endif()
  add_library(lvglplus_renderer STATIC sdk/src/ui.c)
  add_library(LVGLPlus::renderer ALIAS lvglplus_renderer)
  target_link_libraries(lvglplus_renderer PUBLIC lvglplus lvgl)
  add_executable(lvp_runner sdk/ports/runner.c)
  target_link_libraries(lvp_runner PRIVATE lvglplus_renderer lvglplus_package)
  if(EMSCRIPTEN)
    add_executable(lvp_preview sdk/ports/preview.c)
    target_link_libraries(lvp_preview PRIVATE lvglplus_renderer lvglplus_package)
    set_target_properties(lvp_preview PROPERTIES OUTPUT_NAME lvp SUFFIX .js)
    target_link_options(lvp_preview PRIVATE --no-entry
      "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=web,node" "-sALLOW_MEMORY_GROWTH=1"
      "-sEXPORTED_RUNTIME_METHODS=['cwrap','UTF8ToString','FS','HEAPU8','HEAP32']"
      "-sEXPORTED_FUNCTIONS=['_malloc','_free','_lvp_preview_load','_lvp_preview_tick','_lvp_preview_pixels','_lvp_preview_width','_lvp_preview_height','_lvp_preview_error','_lvp_preview_signal_count','_lvp_preview_signal_name','_lvp_preview_signal_type','_lvp_preview_number','_lvp_preview_text','_lvp_preview_quality','_lvp_preview_set','_lvp_preview_set_text','_lvp_preview_begin','_lvp_preview_end','_lvp_preview_disconnect','_lvp_preview_pointer','_lvp_preview_command','_lvp_preview_command_state','_lvp_preview_bounds','_lvp_preview_close']")
  endif()
  if(LVP_UI_DIR)
    add_subdirectory("${LVP_UI_DIR}" "${CMAKE_BINARY_DIR}/generated_ui")
    target_link_libraries(lvp_runner PRIVATE lvglplus_ui)
    target_compile_definitions(lvp_runner PRIVATE LVP_GENERATED LVP_SCHEMA=${LVP_UI_NAME}_schema)
  endif()
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(lvp_can_gateway sdk/ports/socketcan.c)
  target_link_libraries(lvp_can_gateway PRIVATE lvglplus)
  if(LVP_DBC_DIR)
    target_sources(lvp_can_gateway PRIVATE "${LVP_DBC_DIR}/lvp_dbc.c" "${LVP_DBC_DIR}/database.c")
    target_include_directories(lvp_can_gateway PRIVATE "${LVP_DBC_DIR}")
    target_compile_definitions(lvp_can_gateway PRIVATE LVP_WITH_DBC)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  if(LVP_WITH_LVGL AND NOT EMSCRIPTEN)
    add_executable(lvp_ui_lifetime tests/c/ui_lifetime.c)
    target_link_libraries(lvp_ui_lifetime PRIVATE lvglplus_renderer)
    add_test(NAME ui-lifetime COMMAND lvp_ui_lifetime)
  endif()
  add_executable(lvp_core_test tests/c/core_test.c)
  target_link_libraries(lvp_core_test PRIVATE lvglplus lvglplus_package)
  add_test(NAME core COMMAND lvp_core_test)
  if(UNIX AND NOT EMSCRIPTEN AND NOT CMAKE_CROSSCOMPILING)
    find_package(Threads REQUIRED)
    add_executable(lvp_queue_threads tests/c/queue_threads.c)
    target_link_libraries(lvp_queue_threads PRIVATE lvglplus Threads::Threads)
    add_test(NAME threaded-ingress COMMAND lvp_queue_threads)
  endif()
endif()
install(TARGETS lvglplus lvglplus_package ARCHIVE DESTINATION lib)
install(FILES sdk/include/lvglplus.h sdk/include/lvglplus_ui.h DESTINATION include)
